# robots.txt de cualautocompro
#
# Esta lista tiene que decir lo mismo que el `noindex` de cada ruta. Ese
# `noindex` vive en dos archivos: la mayoría se declara ruta por ruta en
# src/app/app.routes.ts, pero el del comparador (/compare y /c/:slug) sale de
# COMPARE_DEFAULT_META, en src/app/core/page-meta.service.ts. Si agregas una
# ruta privada, mira los dos y agrégala acá.

User-agent: *
Allow: /

# Panel interno.
Disallow: /admin

# Cuenta del usuario: comparaciones guardadas, ajustes y recuperar contraseña.
Disallow: /account

# Favoritos: dependen de la sesión, no tienen sentido para un buscador.
Disallow: /favoritos

# El comparador vacío: una pantalla transitoria, no se comparte y el sitio no
# le promete nada al usuario sobre ella. Por eso acá sí bloqueamos el rastreo.
Disallow: /compare

# Los enlaces de comparaciones guardadas (/c/:slug) NO están bloqueados, y esa
# ausencia es deliberada: no la "normalices" con la regla de arriba.
#
# `Disallow` prohíbe rastrear, no indexar. Si bloqueáramos /c/, el crawler
# nunca descargaría la página, nunca ejecutaría el JS y nunca leería el
# `<meta name="robots" content="noindex">` que ese JS inyecta. Bastaría con que
# alguien pegara un enlace en un foro rastreable —y el producto justamente
# empuja a compartirlos— para que Google indexara la URL desnuda, sin snippet.
# Bloquear lograba lo contrario de lo que buscábamos.
#
# Dejando pasar al crawler, este descarga la página, lee el `noindex` y la
# desindexa de verdad. El noindex sale de COMPARE_DEFAULT_META, en
# src/app/core/page-meta.service.ts.

# Autenticación.
Disallow: /login
Disallow: /register
